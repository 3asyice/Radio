   let index = 0;
 const player = document.getElementById("player");
 const episodeList = document.getElementById("episodeList");
 const searchInput = document.getElementById("search");
  const prevBtn = document.getElementById("prevBtn");
 const nextBtn = document.getElementById("nextBtn");
  function loadVideo(i) {
        index = i;
        player.src = playlist[index];
        player.play();
        highlightSelectedEpisode();
      }

      function highlightSelectedEpisode() {
        const items = episodeList.querySelectorAll("li");
        items.forEach((li, idx) => {
          li.style.fontWeight = idx === index ? "bold" : "normal";
        });
      }

      function renderEpisodeList(filter = "") {
        episodeList.innerHTML = "";
        episodeNames.forEach((name, i) => {
          if (name.toLowerCase().includes(filter.toLowerCase())) {
            const li = document.createElement("li");
            li.textContent = name;
            li.style.cursor = "pointer";
            li.onclick = () => loadVideo(i);
            episodeList.appendChild(li);
          }
        });
        highlightSelectedEpisode();
      }

      renderEpisodeList();

      searchInput.addEventListener("input", () => {
        renderEpisodeList(searchInput.value);
      });

      prevBtn.addEventListener("click", () => {
        index = (index - 1 + playlist.length) % playlist.length;
        loadVideo(index);
      });

      nextBtn.addEventListener("click", () => {
        index = (index + 1) % playlist.length;
        loadVideo(index);
      });

      player.addEventListener("ended", () => {
        index = (index + 1) % playlist.length;
        loadVideo(index);
      });

      loadVideo(index);
